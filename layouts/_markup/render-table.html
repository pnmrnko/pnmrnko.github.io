{{- /* Markdown tables, wrapped so that a table wider than the column scrolls sideways on its own (a finger swipe on a phone) instead of widening the page. The wrapper is a labelled, focusable region so the keyboard can scroll it too. Wrapping is decided per column from its content, by line length: a column whose longest cell fits in $measure characters never wraps (names, numbers), a longer one wraps but no narrower than $measure characters, about the shortest line that still reads well in a narrow column. Once the columns can't shrink any further, the table scrolls. */ -}} {{- $measure := 25 }} {{- $longest := dict }} {{- range $part := slice .THead .TBody }} {{- range $part }} {{- range $i, $cell := . }} {{- $n := .Text | plainify | htmlUnescape | countrunes }} {{- $key := string $i }} {{- if gt $n (index $longest $key | default 0) }} {{- $longest = merge $longest (dict $key $n) }} {{- end }} {{- end }} {{- end }} {{- end }} {{- $class := dict }} {{- range $key, $n := $longest }} {{- $class = merge $class (dict $key (cond (le $n $measure) "nowrap" "measure")) }} {{- end }}
