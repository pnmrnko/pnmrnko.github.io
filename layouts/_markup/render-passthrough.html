{{- /* LaTeX → MathML at build time (KaTeX inside Hugo), so pages ship no math JavaScript; the browser lays the MathML out with NewCM Sans Math. A formula that doesn't parse fails the build and names the file. Chrome follows MathML Core and ignores mathvariant (\mathbb, \mathcal, \mathbf …), so those letters are swapped for their Unicode math alphanumerics from data/mathalpha.json (tools/build-mathalpha.py). Below that, the rest of KaTeX's output that Chrome gets wrong is patched: enclosures, table rules, a few stretchy and spacing details. Each fix matches KaTeX's exact markup, so anything else passes through untouched. */ -}} {{- $opts := dict "output" "mathml" "displayMode" (eq .Type "block") "throwOnError" true }} {{- with try (transform.ToMath .Inner $opts) }} {{- with .Err }} {{- errorf "Can't render math in %s: %s" $.Position . }} {{- else }} {{- $html := string .Value }} {{- $re := `<(mi|mn|mtext)([^>]*?) mathvariant="([a-z-]+)"([^>]*)>([^<]*)` }} {{- range findRESubmatch $re $html }} {{- $full := index . 0 }} {{- $tag := index . 1 }} {{- $map := index hugo.Data.mathalpha (index . 3) }} {{- if $map }} {{- $text := "" }} {{- range split (index . 5) "" }} {{- $text = printf "%s%s" $text (index $map . | default .) }} {{- end }} {{- $html = replace $html $full (printf "<%s%s%s>%s" $tag (index . 2) (index . 4) $text $tag) }} {{- end }} {{- end }} {{- /* \boxed, \fbox, \cancel, \sout: no , the stylesheet draws them */}} {{- $html = replaceRE `` `` $html }} {{- $html = replace $html "" "" }} {{- /* \colorbox and \xrightarrow's labels pad with relative widths ("+6pt"), which MathML Core doesn't have; the stylesheet pads instead */}} {{- $html = replaceRE `` `` $html }} {{- $html = replace $html `` `` }} {{- /* \overset{…}{=} and friends come wrapped in an , so they print side by side; unwrap them */}} {{- $html = replaceRE `<(mover|munder)>([^<]*)` `<$1>$2` $html }} {{- $html = replaceRE `` `` $html }} {{- /* mhchem smashes its subscripts and charges (H₂, e⁺) to zero height, and Safari draws a zero-height mpadded as nothing */}} {{- $html = replace $html `X` `X` }} {{- /* vmatrix and Vmatrix bars aren't stretchy in the operator dictionary */}} {{- $html = replaceRE `([∣∥])` `$1` $html }} {{- /* \vec's combining arrow lands left of the letter; use a small → */}} {{- $html = replace $html "\u20d7" `→` }} {{- /* Thin spaces around \cos, \det, \ln … (TeX's \mathop spacing): after the name unless a parenthesis follows (\sin x, \sin(x)), after the power in \cos^2 x, and before the name after a letter (e\cos x) */}} {{- $html = replace $html "\u2061" "\u2061" }} {{- $html = replaceRE "([^<]+)(\u2061)([^<]*)" "$1$3$2" $html }} {{- $html = replaceRE "\u2061(]*>[(\\[])" "\u2061$1" $html }} {{- $html = replaceRE "(|||||)([^<]+]*>\u2061)" "$1$2" $html }} {{- /* A minus or plus right after another operator (=, (, ,) is a sign, as in TeX, not a binary operator with space on both sides */}} {{- $html = replaceRE `(]*>[^<)\]}⟩∣∥|]*)([−+±∓])` `$1$2` $html }} {{- /* array's | and \hline: columnlines and rowlines become classes, e.g. cl1-solid for a rule after the first column */}} {{- range findRE `]*(?:columnlines|rowlines)="[^"]*"[^>]*>` $html }} {{- $tag := . }} {{- $classes := slice }} {{- range $attr, $prefix := dict "columnlines" "cl" "rowlines" "rl" }} {{- range findRESubmatch (printf `%s="([a-z ]+)"` $attr) $tag }} {{- range $i, $line := split (index . 1) " " }} {{- if ne $line "none" }} {{- $classes = $classes | append (printf "%s%d-%s" $prefix (add $i 1) $line) }} {{- end }} {{- end }} {{- end }} {{- end }} {{- with $classes }} {{- $html = replace $html $tag (replace $tag "{{ $html | safeHTML }} {{- else }} {{- $html | safeHTML }} {{- end }} {{- end }} {{- end -}}
