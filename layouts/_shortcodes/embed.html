{{- /* Third-party player in a responsive frame: {{< embed provider="youtube" id="YH3c1QZzRK4" title="…" caption="…" >}} Providers: youtube (no-cookie domain), vimeo (do-not-track), coub, spotify (id is the track id from open.spotify.com/track/…), soundcloud and bandcamp (id is the numeric track id; for soundcloud a full player URL in src works too). link="true" shows a link card instead of a player, for videos whose owner has disabled playback on other sites. ratio="40/17" sets a video's frame to its picture's shape (default 16/9), so a widescreen file isn't letterboxed by the player. */ -}} {{- $provider := .Get "provider" }} {{- $id := .Get "id" }} {{- $src := .Get "src" }} {{- $ratio := true }} {{- $shape := "" }} {{- with .Get "ratio" }} {{- if not (findRE `^\d+(\.\d+)?(/\d+(\.\d+)?)?$` .) }} {{- errorf "embed: ratio %q should look like 40/17 or 2.35 in %s" . $.Position }} {{- end }} {{- $shape = printf "aspect-ratio: %s" (replace . "/" " / ") }} {{- end }} {{- if eq $provider "youtube" }} {{- $src = printf "https://www.youtube-nocookie.com/embed/%s?hl=%s" $id site.Language.Lang }} {{- else if eq $provider "vimeo" }} {{- $src = printf "https://player.vimeo.com/video/%s?dnt=1" $id }} {{- else if eq $provider "coub" }} {{- $src = printf "https://coub.com/embed/%s?muted=false&autostart=false&originalSize=false&startWithHD=true" $id }} {{- else if eq $provider "spotify" }} {{- $src = printf "https://open.spotify.com/embed/track/%s" $id }} {{- $ratio = false }} {{- else if eq $provider "soundcloud" }} {{- with $id }} {{- $src = printf "https://w.soundcloud.com/player/?url=https%%3A//api.soundcloud.com/tracks/%s&color=%%23228be6&auto_play=false&hide_related=true&show_comments=false&show_user=true&show_reposts=false&show_teaser=false&visual=false" . }} {{- end }} {{- $ratio = false }} {{- else if eq $provider "bandcamp" }} {{- $src = printf "https://bandcamp.com/EmbeddedPlayer/track=%s/size=large/bgcol=ffffff/linkcol=228be6/tracklist=false/artwork=small/transparent=true/" $id }} {{- $ratio = false }} {{- else }} {{- errorf "embed: unknown provider %q in %s" $provider .Position }} {{- end }}
